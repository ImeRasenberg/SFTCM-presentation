\begin{frame}{requirements}
	\begin{itemize}
    \item Strongly type–II superconductors\\
     very large Ginzburg–Landau parameter $\kappa \equiv \lambda / \xi >> 1  $
    \item large Maki parameter $\alpha_M$\\
    $\alpha_M = \frac{\sqrt{2}H_{c2}^{orb}}{H_{c2}^P}$\\
	$\alpha_M \geq 1$ limit is Pauli 
    \item  Very clean
    \item Anisotropies of the Fermi–surface and the gap function can stabilize the FFLO state.
  \end{itemize}
\end{frame}

\begin{frame}{Bloch equation}
    \begin{align*}
        \frac{dM_z(t)}{dt} &= \gamma \left(\mathbf{M}(t) \times \mathbf{B}(t)\right)_z - \frac{M_z(t)- M_0}{T_1} \\
        M_z(t) &= M_0 \left(1-e^{-\frac{t}{T_1}} \right)
    \end{align*}
\end{frame}
