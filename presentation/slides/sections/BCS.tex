\begin{frame}{Cooper Pair}

		\only<1->{
			\[
			\mathcal{H} = \sum_{k,\sigma} \epsilon_k \hat{n}_{k,\sigma} + \sum_{k,k'} V_{k,k'} 	\hat{c}_{k\uparrow}^{\dagger}\hat{c}_{-k\downarrow}^{\dagger}\hat{c}_{-k'\downarrow}\hat{c}_{k'\uparrow}
			\]
		}
		\only<2->{
			\[
			\ket{\Psi} = \sum_k a_k \hat{c}_{k\uparrow}^{\dagger}\hat{c}_{-k\downarrow}^{\dagger} \ket{\text{FS}}
			\]
		}
		\only<3->{
			\[
			(E-2\xi_k)a_k = \sum_{k'}V_{k,k'} a_{k'}
			\]
		}
		\only<4->{
			\[
			1 = V\sum_{k}\frac{ 1}{2\xi_k-E} =  V \rho(0) \int_0^{\hbar \omega_D}\frac{ d\xi}{2\xi-E}
			\]
		}
		\only<5->{
			\[
			E = -2\hbar \omega_D e^{-1/(V\rho(0))}
			\]
		}
		
\end{frame}


\begin{frame}{Action}
	
	% Step 1: fermionic action
	\only<1->{
		\[
		S[\psi^*,\psi]
		=
		\int d\tau d^3x
		\left[
		\psi^*_\sigma
		\left(\hbar \partial_\tau +
		\epsilon_k
		-\mu
		\right)
		\psi_\sigma
		-		V\,
		\psi^*_\uparrow
		\psi^*_\downarrow
		\psi_\downarrow
		\psi_\uparrow
		\right]
		\]
	}
	
	% Step 2: Hubbard-Stratonovich
	\only<2->{
		\[
		\psi^*_\uparrow\psi^*_\downarrow\psi_\downarrow\psi_\uparrow
		\;\rightarrow\;
		\frac{|\Delta|^2}{V}
		-
		\Delta\,\psi^*_\uparrow\psi^*_\downarrow
		-
		\Delta^*\,\psi_\downarrow\psi_\uparrow
		\]
		
		\[
		\begin{split}
			\langle\Delta(x,\tau)\rangle_x
			&=
			V\langle
			\psi_\downarrow(x,\tau)\psi_\uparrow(x,\tau)
			\rangle_x\\
			&= V\langle
			\psi_{\downarrow,k}(\tau)\psi_{\uparrow,-k}(\tau)
			\rangle_k
		\end{split}
		\]
		

	}
	
	% Step 3: quadratic effective action
	\only<3->{
		\[
		S_{\mathrm{eff}}
		=
		\int d\tau d^3x
		\left[
		\frac{|\Delta|^2}{V}
		+
		\Psi^\dagger
		\begin{pmatrix}
			i\hbar\omega_n +\epsilon_k - \mu & \Delta \\
			\Delta^* & -i\hbar\omega_n +\epsilon_k - \mu
		\end{pmatrix}
		\Psi
		\right]
		\]
	}
\end{frame}
