	S = S_{e} + S_{e-e} + S_{ph} + S_{e-ph}
	\]
	}
	
	\only<2->{
	\[
	S = S_{e} + S_{e-e} +\sum_{q} (\omega_n^2 + \omega_q^2) \psi_q^\dagger \psi_q + \sum_{q,k,\sigma} g_q c^\dagger_{k+q,\sigma} c_{k,\sigma}(\psi_q + \psi_{-q}^\dagger)
	\]
	}
	
	\only<3->{
	\[
	S = S_{e} + S_{e-e}+ \frac{1}{2}\sum_q \frac{|g_q|^2}{\omega^2 - \omega_q^2} c^\dagger_{k+q,\sigma} c_{k,\sigma}c^\dagger_{k-q,\sigma} c_{k,\sigma}
	\]
	}
	
\end{frame}

\begin{frame}{Band gap}
	\only<1->{
		\[
		S[\psi^*,\psi]
		=
		\sum_{k,\omega_n,\sigma}
		\psi^*_{k\sigma}
		\left(
		-i\hbar\omega_n
		+\epsilon_k
		-\mu
		\right)
		\psi_{k\sigma}
		-
		V
		\sum_{k,k'}
		\psi^*_{k,\uparrow}
		\psi^*_{-k,\downarrow}
		\psi_{-k',\downarrow}
		\psi_{k',\uparrow}
		\]
	}
	
	\only<2->{
		\[
		\Delta \;\rightarrow\; \Delta + V \sum_k \psi_{k,\uparrow}
		\psi_{-k,\downarrow}
		\]
		
		\[
		\Delta = V \sum_k \langle  \psi_{k,\uparrow}
		\psi_{-k,\downarrow} \rangle
		\]
	}
	
	\only<3->{
		\[
		S_{\mathrm{eff}}=\int d\tau d^3x\left[\frac{|\Delta|^2}{V}+\Psi^\dagger
		\begin{pmatrix}
			i\hbar\omega_n +\epsilon_k - \mu & \Delta \\
			\Delta^* & -i\hbar\omega_n +\epsilon_k - \mu
		\end{pmatrix}
		\Psi
		\right]
		\]

	}
\end{frame}
